Type1: equi join 


Step1: Create a Table name  Personal !


CREATE TABLE personal (
    ID INT PRIMARY KEY,
    NAME VARCHAR(50),
    ADDRESS VARCHAR(50),
    CONTACT VARCHAR(15)
);

step2: Create Another Table named  professional


CREATE TABLE professional (
    WORK VARCHAR(50),
    DEPT VARCHAR(50),
    ID INT
);

Step3: Apply Equi join --

SELECT name, address, work 
FROM personal, professional 
WHERE personal.id = professional.id;


--output

+--------+---------+---------+
| ankush | kapali  | teacher |
| altaf  | sakchi  | doctor  |

--OR If We want some more columns then we can also write 

SELECT name, address, contact, work, dept
